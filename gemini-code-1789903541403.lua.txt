local EXPIRE_TIME = 7 * 3600 -- 7 Hours

-- Storage Functions
local function SaveKey(key, filename)
    if writefile then
        pcall(function()
            local data = { key = key, time = os.time() }
            writefile(filename or KEY_FILE, HttpService:JSONEncode(data))
        end)
    end
end

-- FIX 1: Memperbaiki JSONDecode (sebelumnya JSONEncode) agar data ter-load dengan benar
local function GetSavedKeyData(filename)
    local targetFile = filename or KEY_FILE
    if isfile and readfile and isfile(targetFile) then
        local success, result = pcall(function()
            return HttpService:JSONDecode(readfile(targetFile))
        end)
        if success and type(result) == "table" then return result end
    end
    return nil
end

local function IsKeyValid(keyData)
    if not keyData or type(keyData) ~= "table" then return false end
    local key = keyData.key or ""
    local timestamp = keyData.time or 0
    
    local isMichelKey = string.sub(key, 1, 7) == "MICHEL-" and #key >= 15
    local isNotExpired = (os.time() - timestamp) < EXPIRE_TIME or string.find(key, "LIFETIME")
    
    return isMichelKey and isNotExpired
end

local function LoadMainScript()
    task.wait(0.5)
    loadstring(game:HttpGet('https://raw.githubusercontent.com/MichelLyJow/LibraryScriptMichel/refs/heads/main/script.lua'))()
end

local function LoadPremiumScript()
    task.wait(0.5)
    loadstring(game:HttpGet(PREMIUM_RAW_URL))()
end

-- Premium Key Online Verification Function
local function VerifyPremiumKeyOnline(inputKey)
    local success, response = pcall(function()
        return game:HttpGet(PREMIUM_RAW_URL)
    end)
    
    if success and response then
        -- Cek jika input key cocok dengan data di Pastefy
        if string.find(response, inputKey, 1, true) or string.find(inputKey, "LIFETIME") then
            return true
        end
    end
    return false
end

-- 1. AUTO LOAD PREMIUM (FIXED)
local savedPremiumData = GetSavedKeyData(PREMIUM_KEY_FILE)
if savedPremiumData and savedPremiumData.key then
    if VerifyPremiumKeyOnline(savedPremiumData.key) then
        LoadPremiumScript()
        return
    else
        if delfile then pcall(delfile, PREMIUM_KEY_FILE) end
    end
end

-- 2. AUTO LOAD STANDARD KEY

end

local function corner(inst, radius)
    local c = Instance.new("UICorner", inst)
    c.CornerRadius = UDim.new(0, radius or 12)
    return c
end

local function stroke(inst, color, thickness, transparency)
    local s = Instance.new("UIStroke", inst)
    s.Color = color or COL_GLASS_STROKE
    s.Thickness = thickness or 1.2
    s.Transparency = transparency or 0.6
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    return s
end

local function glassGradient(inst, transStart, transEnd)
    local g = Instance.new("UIGradient", inst)
    g.Color = ColorSequence.new(COL_GLASS_BG)
    g.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0, transStart or 0.75),
        NumberSequenceKeypoint.new(1, transEnd or 0.9)
    })
    g.Rotation = 45
    return g
end

-- Execute Logic
ApplyBtn.MouseButton1Click:Connect(function()
    local key = KeyInput.Text

    if key == "" then
        ShowNotification("Key cannot be empty!", "error")
        TweenService:Create(InputStroke, TweenInfo.new(0.15), {Color = COL_ERROR, Transparency = 0.2}):Play()
        return
    end

    if isPremiumMode then
        -- Mode Premium
        ShowNotification("Verifying Premium Code...", "premium")
        if VerifyPremiumKeyOnline(key) then
            SaveKey(key, PREMIUM_KEY_FILE)
            TweenService:Create(InputStroke, TweenInfo.new(0.15), {Color = COL_GOLD, Transparency = 0.2}):Play()
            ShowNotification("Premium Verified! Loading Script...", "success")
            task.wait(1)
            ScreenGui:Destroy()
            LoadPremiumScript()
        else
            TweenService:Create(InputStroke, TweenInfo.new(0.15), {Color = COL_ERROR, Transparency = 0.2}):Play()
            ShowNotification("Invalid or Expired Premium Code!", "error")
        end
    else
        -- Standard Mode / Key Lifetime dari Web
        if string.sub(key, 1, 7) == "MICHEL-" and #key >= 15 then
            if string.find(key, "LIFETIME") then
                -- Jika memasukkan key Lifetime di mode Standard, langsung simpan sebagai Premium
                SaveKey(key, PREMIUM_KEY_FILE)
            else
                SaveKey(key, KEY_FILE)
            end
            
            TweenService:Create(InputStroke, TweenInfo.new(0.15), {Color = COL_SUCCESS, Transparency = 0.2}):Play()
            ShowNotification("Key Verified! Loading Script...", "success")
            task.wait(1)
            ScreenGui:Destroy()
            
            if string.find(key, "LIFETIME") then
                LoadPremiumScript()
            else
                LoadMainScript()
            end
        else
            TweenService:Create(InputStroke, TweenInfo.new(0.15), {Color = COL_ERROR, Transparency = 0.2}):Play()
            ShowNotification("Invalid or Expired Key!", "error")
        end
    end
end)
